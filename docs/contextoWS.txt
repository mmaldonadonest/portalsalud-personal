declare 

kjson clob;
l_response_body_clob  clob ;
l_response_length     NUMBER;
l_response_buffer     VARCHAR2(32767);
l_response_chunk_size BINARY_INTEGER := 32000;
l_offset              INTEGER := 1; 
j apex_json.t_values;
kclob clob;
kapp number;
kmov varchar2(50);
kmenu number;
kexiste number;
knombre varchar2(100);
krol number;
nerror number;
kmessage varchar2(300);
kservicio varchar2(100);
kmas number;
DQL VARCHAR2(1000);


begin 

kclob:= utl_raw.cast_to_varchar2(:body);
apex_json.parse(j,kclob);

 kapp  := apex_json.get_number(p_path=>'id_app',p_values=>j);
 kmenu  := apex_json.get_number(p_path=>'id_menu',p_values=>j);
 knombre := apex_json.get_varchar2(p_path=>'nombre_menu',p_values=>j);
 kmov  := apex_json.get_varchar2(p_path=>'tipo_mov',p_values=>j);
 krol  := apex_json.get_number(p_path=>'id_rol',p_values=>j);
 kservicio  := apex_json.get_varchar2(p_path=>'servicio',p_values=>j);

select upper(kservicio) into kservicio from dual;
select upper(kmov) into kmov from dual;
select upper(kNOMBRE) into kNOMBRE from dual;

--VALIDAMOS 

IF KMOV IS NOT NULL AND KNOMBRE IS NOT NULL THEN 

  if KSERVICIO = 'ROL' THEN --PROCESO DE ROLES
    IF KNOMBRE IS NOT NULL and kapp is not null and krol is not null THEN 
      SELECT COUNT(*) INTO KEXISTE from tbl_apps_rol WHERE ID_APP=KAPP and id_rol=krol;
       if kexiste = 0 then 
        if kmov='ALTA' THEN 
         insert into tbl_apps_rol (id_app,nombre_rol,id_rol) values (kapp,knombre,krol);
          nerror := 0;
        else
         kmessage:= 'la clave app se encuentra registrada en base de datos'; 
         NERROR :=1;
        end if;
      elsif kmov='BAJA' THEN
        DELETE tbl_apps_rol WHERE id_app=KAPP AND ID_ROL=KROL;
        nerror := 0;

      elsif kmov='CAMBIO' THEN
        UPDATE tbl_apps_rol SET nombre_rol=KNOMBRE WHERE id_app=KAPP AND ID_ROL=KROL;
        nerror := 0;
      end if;
    else
       kmessage:= 'El Nombre o la clave app se encuentra null validar datos'; 
       NERROR :=1;
    end if;

  ELSIF KSERVICIO = 'APP' THEN --ALTA DE APP EN LA TABLA
    IF KNOMBRE IS NOT NULL and  kapp is not null THEN 

      SELECT COUNT(*) INTO KEXISTE from tbl_apps_onest WHERE id_app=kapp;

     IF KEXISTE = 0 THEN --NO EXISTE LA APP
       if kmov='ALTA' THEN 
          INSERT INTO tbl_apps_onest (id_app,NOMBRE_APLICACION) VALUES (Kapp,KNOMBRE);
                NERROR := 0;
                kmessage  := 'La aplicación se inserto de forma correcta ';
                NERROR := 0;    
      ELSE
          kmessage  := 'El id de aplicacion ya se encuentra registrada';
            NERROR := 1;
      end if;  
     elsif kmov='BAJA' THEN
       DELETE tbl_apps_onest WHERE ID_APP=KAPP;
       NERROR := 0;   
       kmessage  := 'La aplicación se dío de baja de forma correcta ';
     ELSIF KMOV='CAMBIO' THEN 
       update tbl_apps_onest  set nombre_aplicacion=knombre where id_APP=kapp;
        NERROR := 0;   
       kmessage  := 'La aplicación se cambio de forma correcta ';
     END IF;
    else
       kmessage:= 'El id o nombre de la APP se encuentra null'; 
       NERROR :=1;
    end if;  


  ELSIF KSERVICIO = 'MENU' THEN 

   if knombre is not null and kapp is not null and kmenu is not null and krol is not null then

    --bucamos si existe la app
    SELECT COUNT(*) INTO KEXISTE from tbl_apps_onest WHERE id_app=kapp;
    if kexiste > 0 then 

      SELECT COUNT(*) INTO KEXISTE FROM TBL_APPS_ROL_MENU WHERE ID_APP=KAPP AND ID_MENU=KMENU AND ID_ROL=krol;

     IF KEXISTE = 0 THEN 

      if kmov='ALTA' THEN 

       DQL := 'insert into TBL_APPS_ROL_MENU (id_app,id_rol,id_menu,nombre_menu,estado) 
           values ('||kapp||','||''''||krol||''''||','||kMENU||','||''''||KNOMBRE||''''||',0)';


           execute immediate dql;
            nerror := 0;
      end if;
     else
      if kmov='BAJA' THEN 
      DELETE TBL_APPS_ROL_MENU where id_menu=kmenu and id_rol=krol and id_app=kapp;
        nerror := 0;
      elsif kmov='CAMBIO' THEN 
      update TBL_APPS_ROL_MENU set nombre_menu=knombre where id_menu=kmenu and id_rol=krol and id_app=kapp;    
        nerror := 0;
      else
       nerror := 1;
       kmessage := 'La alta del registro no se puede generar por que ya se encuentra registrada en base de datos';
      end if;         
     END IF; 

    else
      nerror := 1;
      kmessage := 'el id_app no se encuentra registrada en el catalogo de APPS';
    end if; 
   else
      nerror := 1;
      kmessage := 'el id_app,id_menu o id_rol se encuentran null';
   end if; 
  END IF; 



ELSE
    kmessage := 'Algun parametro de ingreso se encuentra vacio';
